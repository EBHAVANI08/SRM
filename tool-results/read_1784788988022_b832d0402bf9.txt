   117→// ============ AI Timetable Generator ============
   118→function generateTimetable(grade: string, section: string, teachers: Teacher[]): Timetable {
   119→  const gradeTeachers = teachers.filter((t) => t.grades.includes(grade))
   120→  const subjectSlots: { subject: string; teacher: Teacher; priority: number }[] = []
   121→
   122→  // Assign teachers to subjects for this grade
   123→  for (const subject of SUBJECTS) {
   124→    const teacher = gradeTeachers.find((t) => t.subjects.includes(subject.name))
   125→    if (teacher) {
   126→      // Core subjects get more periods per week
   127→      const periodsPerWeek = subject.priority === 1 ? 6 : subject.priority === 2 ? 4 : 2
   128→      for (let i = 0; i < periodsPerWeek; i++) {
   129→        subjectSlots.push({ subject: subject.name, teacher, priority: subject.priority })
   130→      }
   131→    }
   132→  }
   133→
   134→  // Shuffle but keep priority subjects in morning slots
   135→  subjectSlots.sort((a, b) => a.priority - b.priority)
   136→
   137→  const days: DayTimetable[] = DAYS.map((day, dayIdx) => {
   138→    const periods: Period[] = PERIOD_TIMES.map((pt, ptIdx) => {
   139→      if (pt.isBreak) {
   140→        return { id: `${day}-break-${ptIdx}`, periodNo: 0, time: pt.time, subject: pt.label || 'Break', teacherId: '', teacherName: '', topic: '', isBreak: true }
   141→      }
   142→      const slotIdx = (dayIdx * 6 + ptIdx) % subjectSlots.length
   143→      const slot = subjectSlots[slotIdx]
   144→      if (!slot) return { id: `${day}-free-${ptIdx}`, periodNo: pt.no, time: pt.time, subject: 'Free', teacherId: '', teacherName: '', topic: '' }
   145→      const topics: Record<string, string> = {
   146→        'Mathematics': 'Algebra — Linear Equations',
   147→        'Science': 'Photosynthesis & Plant Biology',
   148→        'English': 'Essay Writing — Persuasive Texts',
   149→        'Social Studies': 'Ancient Civilizations',
   150→        'Hindi': 'काव्य — कबीर के दोहे',
   151→        'Computer Science': 'Python — Loops & Functions',
   152→        'Physics': 'Motion — Newton\'s Laws',
   153→        'Physical Education': 'Cricket — Batting Technique',
   154→        'Art': 'Watercolor Techniques',
   155→        'Music': 'Raga — Basic Scales',
   156→      }
   157→      return {
   158→        id: `${day}-p${pt.no}`,
   159→        periodNo: pt.no,
   160→        time: pt.time,
   161→        subject: slot.subject,
   162→        teacherId: slot.teacher.id,
   163→        teacherName: slot.teacher.name,
   164→        topic: topics[slot.subject] || 'TBD',
   165→      }
   166→    })
   167→    return { day, periods }
   168→  })
   169→
   170→  return { grade, section, days }
   171→}

   950→// ============ Difficulty Filter ============
   951→// Each question is tagged with an assumed difficulty band based on its position.
   952→// For demo purposes, we tag using a simple rotation — Mixed picks all.
   953→function filterByDifficulty<T>(arr: T[], difficulty: string): T[] {
   954→  if (difficulty === 'Mixed') return arr
   955→  // Easy: pick first half (assumed easier); Medium: middle; Hard: last half
   956→  const n = arr.length
   957→  if (n <= 2) return arr
   958→  if (difficulty === 'Easy') return arr.slice(0, Math.ceil(n / 3) + 1)
   959→  if (difficulty === 'Hard') return arr.slice(Math.floor(n / 3) * 2)
   960→  return arr.slice(Math.floor(n / 3), Math.floor(n / 3) * 2 + 1)
   961→}

   963→// ============ Paper Generator ============
   964→function generatePaper(config: PaperConfig): GeneratedPaper {
   965→  const bank = QUESTION_BANK[config.subject] || FALLBACK_BANK
   966→
   967→  // Section marks distribution (percentages of total)
   968→  // A: MCQ       — 20% (1 mark each)
   969→  // B: Short     — 30% (2 marks each)
   970→  // C: Long      — 30% (3-4 marks each)
   971→  // D: Case Study— 20% (4 marks each, with sub-questions of 1 mark)
   972→  const sections: GeneratedSection[] = []
   973→  const types = config.questionTypes
   974→
   975→  // Helper: compute marks-per-section from a percentage
   976→  const marksFor = (pct: number) => Math.max(1, Math.round((config.totalMarks * pct) / 100))
   977→
   978→  // ----- Section A: MCQ (1 mark each) -----
   979→  if (types.includes('MCQ')) {
   980→    const sm = marksFor(20)
   981→    const pool = filterByDifficulty(bank.mcq, config.difficulty)
   982→    const need = Math.min(sm, pool.length)
   983→    const picked = pool.slice(0, need)
   984→    sections.push({
   985→      label: 'A',
   986→      type: 'MCQ',
   987→      instruction: `Multiple Choice Questions. Each question carries 1 mark. Attempt all questions. Choose the correct option.`,
   988→      questions: picked.map((m, i) => ({
   989→        number: String(i + 1),
   990→        text: m.text,
   991→        options: m.options,
   992→        marks: 1,
   993→        answerLines: 1,
   994→      })),
   995→      totalMarks: need,
   996→    })
   997→  }
   998→
   999→  // ----- Section B: Short Answer (2 marks each) -----
  1000→  if (types.includes('Short Answer')) {
  1001→    const target = marksFor(30)
  1002→    const perQ = 2
  1003→    const count = Math.min(Math.floor(target / perQ), bank.short.length)
  1004→    const pool = filterByDifficulty(bank.short, config.difficulty)
  1005→    const picked = pool.slice(0, count)
  1006→    sections.push({
  1007→      label: 'B',
  1008→      type: 'Short Answer',
  1009→      instruction: `Very Short / Short Answer Questions. Each question carries ${perQ} marks. Attempt all questions.`,
  1010→      questions: picked.map((s, i) => ({
  1011→        number: String(i + 1),
  1012→        text: s.text,
  1013→        marks: perQ,
  1014→        answerLines: 4,
  1015→      })),
  1016→      totalMarks: count * perQ,
  1017→    })
  1018→  }
  1019→
  1020→  // ----- Section C: Long Answer (4 marks each) -----
  1021→  if (types.includes('Long Answer')) {

   842→                  <span className={`w-1.5 h-1.5 rounded-full ${l.status === 'SUCCESS' ? 'bg-emerald-500' : 'bg-rose-500'}`} />
   843→                  <span className="text-slate-700 font-medium">{l.date}</span>
   844→                  <span className="text-slate-500">{l.rowsSynced} rows</span>
   845→                  {l.error && <span className="text-rose-600 truncate max-w-xs">· {l.error}</span>}

   849→            ))}
   850→          </div>
   851→        </div>
   852→      )}
   853→    </Card>
   854→  )

   857→function GoogleSheetConfigForm({ config, onSaved }: { config: any; onSaved: () => void }) {
   858→  const [spreadsheetId, setSpreadsheetId] = useState(config?.spreadsheetId || '')
   859→  const [sheetName, setSheetName] = useState(config?.sheetName || 'Daily Attendance')
   860→  const [serviceAccountEmail, setServiceAccountEmail] = useState(config?.serviceAccountEmail || '')
   861→  const [privateKey, setPrivateKey] = useState('')
   862→  const [apiKey, setApiKey] = useState('')
   863→  const [syncTime, setSyncTime] = useState(config?.syncTime || '18:00')
   864→  const [isActive, setIsActive] = useState(config?.isActive ?? true)
   865→  const [saving, setSaving] = useState(false)
   866→
   867→  const handleSave = async () => {
   868→    if (!spreadsheetId) { toast.error('Spreadsheet ID is required'); return }
   869→    setSaving(true)
   870→    try {
   871→      const { apiFetch } = await import('@/lib/apiFetch')
   872→      const res = await apiFetch('/api/attendance/google-sheet-sync/config', {
   873→        method: 'PUT',
   874→        body: JSON.stringify({
   875→          spreadsheetId,
   876→          sheetName,
   877→          serviceAccountEmail: serviceAccountEmail || undefined,
   878→          privateKey: privateKey || undefined,
   879→          apiKey: apiKey || undefined,
   880→          syncTime,
   881→          isActive,
   882→        }),
   883→      })
   884→      const data = await res.json()
   885→      if (data.success) {
   886→        toast.success('Google Sheet sync configured')
   887→        onSaved()
   888→      } else {
   889→        toast.error(`Save failed: ${data.error}`)
   890→      }
   891→    } catch (e: any) {
   892→      toast.error(`Error: ${e?.message}`)
   893→    } finally {
   894→      setSaving(false)
   895→    }
   896→  }
   897→
   898→  return (
   899→    <div className="p-3 rounded-xl bg-slate-50 border border-slate-200 space-y-2.5 mb-2">
   900→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">

   902→          <label className="text-[10px] font-semibold text-slate-600 block mb-0.5">Google Spreadsheet ID *</label>
   903→          <Input value={spreadsheetId} onChange={(e) => setSpreadsheetId(e.target.value)} placeholder="1ABC...XYZ (from the sheet URL)" className="h-8 text-[11px] rounded-lg bg-white" />
   904→          <p className="text-[9px] text-slate-400 mt-0.5">From docs.google.com/spreadsheets/d/<b>THIS_PART</b>/edit</p>
   905→        </div>
   906→        <div>
   907→          <label className="text-[10px] font-semibold text-slate-600 block mb-0.5">Sheet Tab Name</label>
   908→          <Input value={sheetName} onChange={(e) => setSheetName(e.target.value)} className="h-8 text-[11px] rounded-lg bg-white" />
   909→        </div>
   910→        <div>
   911→          <label className="text-[10px] font-semibold text-slate-600 block mb-0.5">Daily Sync Time (24h)</label>
   912→          <Input type="time" value={syncTime} onChange={(e) => setSyncTime(e.target.value)} className="h-8 text-[11px] rounded-lg bg-white" />
   913→        </div>
   914→        <div>
   915→          <label className="text-[10px] font-semibold text-slate-600 block mb-0.5">Active</label>
   916→          <div className="flex items-center gap-2 h-8">
   917→            <Switch checked={isActive} onCheckedChange={setIsActive} />
   918→            <span className="text-[10px] text-slate-600">{isActive ? 'Auto-sync enabled' : 'Paused'}</span>
   919→          </div>
   920→        </div>
   921→      </div>
   922→      <div className="pt-2 border-t border-slate-200">
   923→        <div className="text-[10px] font-semibold text-slate-600 mb-1.5">Authentication (choose one):</div>
   924→        <div className="space-y-2">
   925→          <div>
   926→            <label className="text-[10px] text-slate-500 block mb-0.5">Service Account Email (recommended for production)</label>
   927→            <Input value={serviceAccountEmail} onChange={(e) => setServiceAccountEmail(e.target.value)} placeholder="learnx-sync@project.iam.gserviceaccount.com" className="h-8 text-[11px] rounded-lg bg-white" />
   928→          </div>
   929→          <div>
   930→            <label className="text-[10px] text-slate-500 block mb-0.5">Service Account Private Key {config?.privateKeyEnc === '***CONFIGURED***' && '(configured — leave blank to keep)'}</label>
   931→            <Textarea value={privateKey} onChange={(e) => setPrivateKey(e.target.value)} placeholder="-----BEGIN PRIVATE KEY-----&#10;...&#10;-----END PRIVATE KEY-----" className="text-[10px] rounded-lg bg-white font-mono min-h-[50px]" />
   932→          </div>
   933→          <div>
   934→            <label className="text-[10px] text-slate-500 block mb-0.5">OR API Key (simpler, for public sheets) {config?.apiKey === '***CONFIGURED***' && '(configured — leave blank to keep)'}</label>
   935→            <Input value={apiKey} onChange={(e) => setApiKey(e.target.value)} placeholder="AIzaSy..." className="h-8 text-[11px] rounded-lg bg-white font-mono" />
   936→          </div>

   938→      </div>

   942→          {saving ? <RefreshCw className="w-3.5 h-3.5 animate-spin" /> : <CheckCircle2 className="w-3.5 h-3.5" />}

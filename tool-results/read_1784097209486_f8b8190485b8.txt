   454→/**

   456→ * (attendance, fees, exams, behavior, health, transport, hostel, activities,
   457→ * documents, report cards, PTM meetings), auto-detects siblings via
   458→ * guardianPhone match, surfaces an AI at-risk summary, and exposes
   459→ * "Message Parent" + "Inform Class Teacher" actions.
   460→ */
   461→function SISBiodataModal({
   462→  row,
   463→  messageText,
   464→  setMessageText,
   465→  onClose,
   466→  onMessageParent,
   467→  onInformTeacher,
   468→}: {
   469→  row: typeof SIS_STUDENTS[0]
   470→  messageText: string
   471→  setMessageText: (s: string) => void
   472→  onClose: () => void
   473→  onMessageParent: (body: string, subject: string) => void
   474→  onInformTeacher: (teacherName: string, teacherPhone: string, body: string, subject: string) => void
   475→}) {
   476→  // Resolve the full Student record (with linked records) from school-data.ts
   477→  const fullId = SIS_TO_FULL_STUDENT_ID[row.id]
   478→  const full = fullId ? STUDENTS.find((s) => s.id === fullId) : undefined
   479→
   480→  // Sibling detection — match by guardianPhone (excluding self)
   481→  const siblings = full
   482→    ? STUDENTS.filter((s) => s.id !== full.id && s.guardianPhone === full.guardianPhone)
   483→    : []
   484→
   485→  // Resolve class teacher for the student's grade
   486→  const gradeNum = row.grade.split('-')[0]
   487→  const teacherKey = `Grade ${gradeNum}`
   488→  const teacher = GRADE_TEACHERS_LOOKUP[teacherKey] || GRADE_TEACHERS_LOOKUP['Grade 1']
   489→
   490→  // AI at-risk summary (rule-based, deterministic, explainable)
   491→  const attendancePct = full && full.attendance.length > 0
   492→    ? Math.round(full.attendance.filter((a) => a.status === 'PRESENT').length / full.attendance.length * 100)
   493→    : parseInt(row.attendance)
   494→  const feesOverdue = full ? full.fees.some((f) => f.status === 'OVERDUE' || f.status === 'PENDING') : row.fees !== 'Paid'
   495→  const negativeBehavior = full ? full.behaviorRecords.filter((b) => b.type === 'NEGATIVE').length : 0
   496→  const positiveBehavior = full ? full.behaviorRecords.filter((b) => b.type === 'POSITIVE').length : 0
   497→  const atRisk = attendancePct < 85 || feesOverdue || negativeBehavior > positiveBehavior + 2
   498→
   499→  const aiSummary = atRisk
   500→    ? `⚠️ At-Risk: Attendance ${attendancePct}%, fees ${feesOverdue ? 'overdue' : 'ok'}, ${negativeBehavior} negative behavior incidents vs ${positiveBehavior} positive. Recommend counselor intervention within 7 days.`
   501→    : `✅ Healthy: Attendance ${attendancePct}%, fees ${feesOverdue ? 'overdue' : 'current'}, ${positiveBehavior} positive behavior incidents. Continue current engagement.`
   502→
   503→  const handleSendParent = () => {
   504→    const subject = `Message from LearnX SIS regarding ${row.name}`
   505→    const body = messageText.trim()
   506→      ? `Dear ${row.parent},\n\n${messageText}\n\n— LearnX SIS (regarding ${row.name}, Grade ${row.grade})`
   507→      : `Dear ${row.parent},\n\nThis is a message from the LearnX Student Information System regarding ${row.name} (Grade ${row.grade}). Please find the latest profile snapshot below:\n\nAttendance: ${attendancePct}%\nFees: ${feesOverdue ? 'OVERDUE — please clear at the earliest' : 'Up to date'}\nBehavior: ${positiveBehavior} positive / ${negativeBehavior} negative incidents this term\n\n— LearnX SIS`
   508→    onMessageParent(body, subject)
   509→  }
   510→
   511→  const handleInformTeacher = () => {
   512→    const subject = `SIS Note regarding ${row.name} (Grade ${row.grade})`
   513→    const body = messageText.trim()
   514→      ? `Dear ${teacher.name},\n\nNote from the LearnX SIS regarding ${row.name} (Grade ${row.grade}, Roll ${row.roll}):\n\n${messageText}\n\nStudent summary:\n- Attendance: ${attendancePct}%\n- Fees: ${feesOverdue ? 'Overdue' : 'Current'}\n- Behavior: ${positiveBehavior}+ / ${negativeBehavior}- this term\n- AI Risk: ${atRisk ? 'AT-RISK' : 'Healthy'}\n\nParent / Guardian: ${row.parent} (${row.phone})\n\n— LearnX SIS`
   515→      : `Dear ${teacher.name},\n\nSharing an update regarding ${row.name} (Grade ${row.grade}, Roll ${row.roll}).\n\nStudent summary:\n- Attendance: ${attendancePct}%\n- Fees: ${feesOverdue ? 'Overdue' : 'Current'}\n- Behavior: ${positiveBehavior}+ / ${negativeBehavior}- this term\n- AI Risk: ${atRisk ? 'AT-RISK — please monitor' : 'Healthy'}\n\nParent / Guardian: ${row.parent} (${row.phone})\n\n— LearnX SIS`
   516→    onInformTeacher(teacher.name, teacher.phone, body, subject)
   517→  }
   518→
   519→  return (
   520→    <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }} className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-900/50 backdrop-blur-sm" onClick={onClose}>
   521→      <motion.div initial={{ scale: 0.95, opacity: 0 }} animate={{ scale: 1, opacity: 1 }} exit={{ scale: 0.95, opacity: 0 }} onClick={(e) => e.stopPropagation()} className="bg-white rounded-2xl shadow-2xl max-w-4xl w-full max-h-[92vh] overflow-hidden flex flex-col" style={{ borderTop: '4px solid #0D9488' }}>
   522→        {/* Header */}
